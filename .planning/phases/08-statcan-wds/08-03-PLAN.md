    2. Each tool function follows this pattern:
       ```python
       @tool
       async def sc_tool_name(param: type, lang: Literal["en", "fr"] = "en") -> dict:
           """One-line description.

           Use for: when an agent should use this tool.
           Keywords: statcan statistics canada [domain-specific terms] [8+ keywords]
           """
           try:
               data, was_cached = await client_function(param)
               return make_response(
                   [item.model_dump() for item in data] if isinstance(data, list) else data.model_dump(),
                   api_name=_API_NAME, api_url=_API_URL, cached=was_cached, lang=lang,
               )
           except httpx.HTTPStatusError as exc:
               if exc.response.status_code == 409:
                   return make_error("UPSTREAM_UNAVAILABLE",
                       "StatCan WDS is in its maintenance window (00:00-08:30 EST). Try again after 08:30 EST.",
                       lang=lang)
               return make_error("UPSTREAM_ERROR", str(exc), lang=lang)
           except ValueError as exc:
               return make_error("UPSTREAM_ERROR", str(exc), lang=lang)
           except Exception as exc:
               return make_error("UPSTREAM_ERROR", f"Unexpected error: {exc}", lang=lang)
       ```
