        "quarterly": ["quarterly"],
        "annual": ["annual", "1921"],
        "age_sex": ["age", "sex"],

    async def fetch_air_quality_stations(
        max_records: int = 5000, include_geometry: bool = False,
    ) -> tuple[dict[str, Any], bool]:
        cache_key = f"{CACHE_KEY_PREFIX}geodiscover:aqhi:{max_records}:{include_geometry}"
        limiter = get_limiter(RATE_GROUP_GEODISCOVER, RATE_LIMIT_GEODISCOVER)

        async def _fetch():
            await limiter.acquire()
            features, truncated = await arcgis_hub.query_feature_service(
                AQHI_AIR_LAYER_URL, AQHI_STATIONS_LAYER_ID,
                where=None, out_fields="*",
                include_geometry=include_geometry, max_records=max_records,
            )
            return {"stations": features, "count": len(features), "truncated": truncated}

        return await cached_fetch(cache_key, CACHE_TTL_LIVE, _fetch)


    async def fetch_water_advisories(
        advisory_type: Literal["river", "water_management", "drought", "ice_cover", "water_sharing"],
        max_records: int = 5000, include_geometry: bool = False,
    ) -> tuple[dict[str, Any], bool]:
        if advisory_type not in WATER_ADVISORY_LAYERS:
            raise ValueError(f"advisory_type must be one of {list(WATER_ADVISORY_LAYERS)}, got '{advisory_type}'")
        layer_id = WATER_ADVISORY_LAYERS[advisory_type]
        cache_key = f"{CACHE_KEY_PREFIX}geodiscover:water:{advisory_type}:{max_records}:{include_geometry}"
        limiter = get_limiter(RATE_GROUP_GEODISCOVER, RATE_LIMIT_GEODISCOVER)

        async def _fetch():
            await limiter.acquire()
            features, truncated = await arcgis_hub.query_feature_service(
                RIVER_FORECAST_FS_URL, layer_id,
                where=None, out_fields="*",
                include_geometry=include_geometry, max_records=max_records,
            )
            return {"advisories": features, "count": len(features), "truncated": truncated,
                    "advisory_type": advisory_type}

        return await cached_fetch(cache_key, CACHE_TTL_LIVE, _fetch)


    async def fetch_crop_production() -> tuple[dict[str, Any], bool]:
        """Historical major crop production from CKAN (2000-2014 official statistic)."""
        cache_key = f"{CACHE_KEY_PREFIX}ckan:crop_production"
        limiter = get_limiter(RATE_GROUP_CKAN, RATE_LIMIT_CKAN)

        async def _fetch():
            await limiter.acquire()
            pkg, _ = await fetch_dataset_details("major-crop-production-alberta")
            csv_resource = next(
                (r for r in pkg.get("resources", []) if (r.get("format") or "").upper() == "CSV"),
                None,
            )
            if not csv_resource:
                return {"rows": [], "count": 0, "note": "No CSV resource found in package"}
            rows, _ = await fetch_and_parse(csv_resource["url"], ttl=CACHE_TTL_ANNUAL)
            return {"rows": rows, "count": len(rows), "source_url": csv_resource["url"]}

        return await cached_fetch(cache_key, CACHE_TTL_ANNUAL, _fetch)


    async def fetch_population_estimates(
        breakdown: Literal["csd", "quarterly", "annual", "age_sex", "sub_provincial", "components_of_growth"] = "csd",
    ) -> tuple[dict[str, Any], bool]:
        if breakdown not in POPULATION_BREAKDOWN_HINTS:
            raise ValueError(f"breakdown must be one of {list(POPULATION_BREAKDOWN_HINTS)}, got '{breakdown}'")
        cache_key = f"{CACHE_KEY_PREFIX}ckan:population:{breakdown}"
        limiter = get_limiter(RATE_GROUP_CKAN, RATE_LIMIT_CKAN)
        hints = POPULATION_BREAKDOWN_HINTS[breakdown]

        async def _fetch():
            await limiter.acquire()
            pkg, _ = await fetch_dataset_details("alberta-population-estimates-data-tables")
            xlsx_resource = next(
                (r for r in pkg.get("resources", [])
                 if (r.get("format") or "").upper() in ("XLSX", "XLS")
                 and any(h in (r.get("url") or r.get("name") or "").lower() for h in hints)),
                None,
            )
            if not xlsx_resource:
                # Fallback: any XLSX in package
                xlsx_resource = next(
                    (r for r in pkg.get("resources", []) if (r.get("format") or "").upper() in ("XLSX", "XLS")),
                    None,
                )
            if not xlsx_resource:
                return {"rows": [], "count": 0, "note": "No XLSX resource found in package", "breakdown": breakdown}
            rows, _ = await fetch_and_parse(xlsx_resource["url"], ttl=CACHE_TTL_ANNUAL)
            return {"rows": rows, "count": len(rows), "source_url": xlsx_resource["url"], "breakdown": breakdown}

        return await cached_fetch(cache_key, CACHE_TTL_ANNUAL, _fetch)


    async def fetch_provincial_parks(

            )
            return {"parks": features, "count": len(features), "truncated": truncated}

        return await cached_fetch(cache_key, CACHE_TTL_STATIC, _fetch)
