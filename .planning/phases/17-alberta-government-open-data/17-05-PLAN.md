<objective>
Implement the 3 AHS (Alberta Health Services) tools that surface Alberta's health-facility data — hospitals (101 with IP/ED flags), AHS zones (5 with population), and EMS stations + PCN clinics (dispatched). All use AHSGIS ArcGIS Online org (NOT GeoDiscover health folder, token-walled per Pitfall 3). ER wait times deferred — AHS publishes via JavaScript widget only (Pitfall 9, Open Question 1); the freed tool slot is used for facility-type dispatch instead.

Purpose: Health-facility data is reference data agents combine with population estimates and AHS zones for service-area analysis. EMS/PCN clinic locations are particularly useful for emergency-response queries.

Output: 3 working AHS tools, each cached 24h (data stability). Optional `zone=` filter on hospitals via name substring match (no polygon containment to avoid additional GeoDiscover round-trip).
</objective>

    ```python
    async def fetch_hospitals(
        zone: str | None = None,
        max_records: int = 5000,
        include_geometry: bool = False,
    ) -> tuple[dict[str, Any], bool]:
        """101 AHS hospitals with IP/ED flags. Optional zone= name-substring filter."""
        cache_key = f"{CACHE_KEY_PREFIX}ahs:hospitals:{zone}:{max_records}:{include_geometry}"
        limiter = get_limiter(RATE_GROUP_AHS, RATE_LIMIT_AHS)

        async def _fetch():
            await limiter.acquire()
            features, truncated = await arcgis_hub.query_feature_service(
                AHS_HOSPITALS_FS_URL, 0,
                where=None, out_fields="*",
                include_geometry=include_geometry, max_records=max_records,
            )
            if zone:
                z = zone.lower()
                features = [f for f in features
                            if z in (f.get("Location", "") or "").lower()
                            or z in (f.get("location", "") or "").lower()]
            return {"features": features, "count": len(features), "truncated": truncated}

        return await cached_fetch(cache_key, CACHE_TTL_STATIC, _fetch)


    async def fetch_ahs_zones(
        include_geometry: bool = False,
    ) -> tuple[dict[str, Any], bool]:
        """5 AHS zones with POP2006/2011/2016."""
        cache_key = f"{CACHE_KEY_PREFIX}ahs:zones:{include_geometry}"
        limiter = get_limiter(RATE_GROUP_AHS, RATE_LIMIT_AHS)

        async def _fetch():
            await limiter.acquire()
            features, truncated = await arcgis_hub.query_feature_service(
                AHS_ZONE_FS_URL, 0,
                where=None, out_fields="*",
                include_geometry=include_geometry, max_records=10,
            )
            # Normalize population field names
            normalized = []
            for f in features:
                normalized.append({
                    "zone_name": f.get("Zone_Name") or f.get("zone_name"),
                    "zone_id": f.get("Zone_ID") or f.get("zone_id"),
                    "pop_2006": f.get("POP2006") or f.get("pop_2006"),
                    "pop_2011": f.get("POP2011") or f.get("pop_2011"),
                    "pop_2016": f.get("POP2016") or f.get("pop_2016"),
                    **({"geometry": f.get("geometry")} if include_geometry else {}),
                })
            return {"features": normalized, "count": len(normalized), "truncated": truncated}

        return await cached_fetch(cache_key, CACHE_TTL_STATIC, _fetch)


    async def fetch_health_facilities(
        facility_type: Literal["ems", "pcn_clinic"],
        max_records: int = 5000,
        include_geometry: bool = False,
    ) -> tuple[dict[str, Any], bool]:
        url_map = {"ems": AHS_EMS_FS_URL, "pcn_clinic": PCN_CLINICS_FS_URL}
        if facility_type not in url_map:
            raise ValueError(f"facility_type must be one of {list(url_map)}, got '{facility_type}'")
        url = url_map[facility_type]
        cache_key = f"{CACHE_KEY_PREFIX}ahs:facility:{facility_type}:{max_records}:{include_geometry}"
        limiter = get_limiter(RATE_GROUP_AHS, RATE_LIMIT_AHS)

        async def _fetch():
            await limiter.acquire()
            features, truncated = await arcgis_hub.query_feature_service(
                url, 0, where=None, out_fields="*",
                include_geometry=include_geometry, max_records=max_records,
            )
            return {"features": features, "count": len(features), "truncated": truncated,
                    "facility_type": facility_type}

        return await cached_fetch(cache_key, CACHE_TTL_STATIC, _fetch)
    ```
