       ```python
       async def fetch_permanent_residents(
           breakdown: str = "country",
           lang: str = "en",
       ) -> tuple[list[dict], bool]:
           """Fetch IRCC permanent residents data for given breakdown."""
           registry = DATASET_REGISTRY["pr"]
           if breakdown not in registry:
               valid = sorted(registry.keys())
               raise ValueError(f"Unknown breakdown {breakdown!r}. Valid: {valid}")
           urls = registry[breakdown]
           if lang not in urls:
               raise ValueError(f"Language {lang!r} not available for pr/{breakdown}. Valid: {sorted(urls.keys())}")
           return await fetch_and_parse(urls[lang])
       ```
