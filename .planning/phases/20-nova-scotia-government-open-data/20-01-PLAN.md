<!-- Prescriptive function surface — from 20-RESEARCH.md "shared/socrata.py Design". -->
async def search_catalog(domain, q="", limit=10, offset=0, only="datasets", *, app_token=None, httpx_client=None) -> dict
  url = f"https://{domain}/api/catalog/v1"
  params = {"domains": domain, "q": q, "limit": limit, "only": only}; if offset>0: params["offset"]=offset
  headers = {"X-App-Token": app_token} if app_token else {}
  Returns the raw catalog JSON {results, resultSetSize, timings, warnings}.

async def get_dataset_metadata(domain, dataset_id, *, app_token=None, httpx_client=None) -> dict
  url = f"https://{domain}/api/views/{dataset_id}.json"
  Returns flat dict: id, name, category, description, columns (list of {name, field_name, data_type, description}),
  attribution, license_name, publication_date, tags.

async def query_dataset(domain, dataset_id, where=None, select=None, order=None, limit=1000, offset=0, q=None, group=None, *, app_token=None, httpx_client=None) -> list[dict]
  url = f"https://{domain}/resource/{dataset_id}.json"
  params = {"$limit": limit}; add $where/$select/$order/$q/$group only when provided; add $offset only when offset>0.
  Returns the flat list of row dicts (no envelope).

def shape_catalog_result(result) -> dict
  Flatten one results[i] entry → {id, name, description, category, tags, department, permalink, updated_at, download_count, type, column_names}.

Constants: DEFAULT_TIMEOUT=30.0, MAX_DESCRIPTION_CHARS=500, CATALOG_PATH, RESOURCE_PATH, VIEWS_PATH.
Pitfall (Pitfall 8): omit $offset / offset when 0.
